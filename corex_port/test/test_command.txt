# Tengine CoreX (ivcore11) FULL test-suite re-run (v3) — device = GPU 1 (CUDA_VISIBLE_DEVICES=1).
# onnx 1.22.0 is installed, so the ONNX node test data is generated and the 75 ONNX op
# ctests are actually executed. Full output -> corex_port/test/test.log.
source corex_port/env/corex_env.sh

# Build test suite + convert_tool (CUDA ON + tests ON). Protobuf 3.21.12 enables onnx op tests.
cmake -S . -B build -DTENGINE_ENABLE_CUDA=ON -DTENGINE_BUILD_TESTS=ON -DTENGINE_BUILD_CONVERT_TOOL=ON -DCMAKE_BUILD_TYPE=Release
cmake --build build -j"$(nproc)" -- -k        # -k: keep going past out-of-scope OpenCV example errors (blocker #4)

# Generate ../onnx_node/<op>/{onnx.tmfile,test_data_set_*/*.pb} for all 75 op tests from the
# installed onnx backend node data, via Tengine's convert_tool (onnx2tengine):
bash corex_port/test/gen_onnx_node_data.sh          # 65/75 convert directly
python3 corex_port/test/regen_hard_ops.py           # 10 opset-25 / missing-attr models rebuilt via onnx.helper

# PART A — full ctest collection (75 ONNX op tests, Tengine CPU REFERENCE backend; not CUDA).
( cd build && ctest --output-on-failure -V --timeout 120 )    # 68 pass, 7 fail (Tengine CPU-backend defects)

# PART B — CUDA-backend evidence: on-GPU CUDA-vs-CPU top-5 parity over 9 benchmark models (CUDA-exercising).
bash corex_port/test/run_tests.sh                   # 9/9 pass
